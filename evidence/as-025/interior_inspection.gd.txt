extends SceneTree
# Inspection camera at standing eye height on existing deck rings.
# This capture proves presentation only, not player traversal.
func _initialize() -> void:
	root.size = Vector2i(648, 557)
	root.content_scale_size = Vector2i(648, 557)
	root.disable_3d = true
	_run.call_deferred()
func _run() -> void:
	var main = load("res://main.tscn").instantiate()
	root.add_child(main)
	main._settings.apply_quality(0)
	main._settings.shadow_quality = 0
	main._settings.msaa = 0
	main._apply_settings()
	await process_frame
	main.set_process(false)
	main._hud.hide()
	main._touch.hide()
	var camera := Camera3D.new()
	camera.fov = 82.0
	camera.far = 500.0
	main.add_child(camera)
	camera.make_current()
	var prefix := ""
	for arg in OS.get_cmdline_user_args():
		if arg.begins_with("--interior-capture="): prefix = arg.trim_prefix("--interior-capture=")
	var views = [
		["service_bay", Vector3(-11,12.52,-171), Vector3(-15.5,16,-165.5)],
		["cable_bay", Vector3(-10,23.52,-172), Vector3(-13,26,-165.6)],
		["shaft", Vector3(0,12.52,-171), Vector3(0,42,-149)]
	]
	for view in views:
		camera.position = view[1]
		camera.look_at(view[2])
		root.disable_3d = false
		await RenderingServer.frame_post_draw
		await RenderingServer.frame_post_draw
		var path = prefix + "_" + view[0] + ".png"
		if root.get_texture().get_image().save_png(path) != OK:
			quit(31)
			return
		print("INTERIOR_CAPTURE ", path)
		root.disable_3d = true
	print("INTERIOR_PRESENTATION_COMPLETE")
	quit(0)
